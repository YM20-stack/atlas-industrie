package cs;

import java.util.Scanner;

class Monstre{
	String TypeMonstre;
	int PointsDeVieMonstre;
	int AttaqueMonstre;
	public Monstre(String MonsterType, int NbPoints, int ATK) {
		TypeMonstre = MonsterType;
		PointsDeVieMonstre = NbPoints;
		AttaqueMonstre = ATK;
	}
	public void setAttaqueMonstre(int UpdatedATK) {
		AttaqueMonstre = UpdatedATK; 
	}
	public int getPointsDeVieMonstre() {
		return PointsDeVieMonstre;
	}
	public void setPointsDeVieMonstre(int UpdatedNbPoints) {
		PointsDeVieMonstre = UpdatedNbPoints;
	}
	public String getTypeMonstre() {
		return TypeMonstre; 
	}
	public void MonstreAttaquer(Joueur J) {
		 int t = J.getPointsDeVieJoueur()-AttaqueMonstre; 
		 if(t>0) {
		 J.setPointsDeVieJoueur(t);
		 }
		 else {
			 J.setPointsDeVieJoueur(0);
		 }
			 
	}
	public void diminuerPointsDeVieMonstre(int i) {
		 int t = this.getPointsDeVieMonstre() - i; 
		 if(t>0)
		 this.setPointsDeVieMonstre(t);
		 else
		 {
			 this.setPointsDeVieMonstre(0); 
		 }
	}
}
class Gobelin extends Monstre{
	public Gobelin() {
		super("Gobelin", 40, 10);
	}
	public void MonstreAttaquer(Joueur J) {
		System.out.println("Le Gobelin vous attaque et inflige 10 dégâts !");
		super.MonstreAttaquer(J); 
		System.out.println("Vos points de vie : "+J.getPointsDeVieJoueur());
	}
}
class Orc extends Monstre{
	public Orc() {
		super("Orc", 60, 15);
	}
	public void MonstreAttaquer(Joueur J) {
		System.out.println("L'Orc vous attaque et inflige 15 dégâts !");
		super.MonstreAttaquer(J);
		System.out.println("Vos points de vie : "+J.getPointsDeVieJoueur());
	}
}
class Dragon extends Monstre{
	private int NumeroTourDragon;
	public Dragon() {
		super("Dragon",100,20);
		NumeroTourDragon = 1;
	}
	public void MonstreAttaquer(Joueur J) {
		if(NumeroTourDragon%3 != 0) {
		System.out.println("Le Dragon vous attaque et inflige 20 dégâts !");
		super.MonstreAttaquer(J);
		NumeroTourDragon += 1;
		System.out.println("Vos points de vie : "+J.getPointsDeVieJoueur());
		}
		else {
			System.out.println("Le Dragon réalise son attaque spéciale !");
			System.out.println("Il vous inflige 30 dégâts !");
			setAttaqueMonstre(30);
			super.MonstreAttaquer(J);
			NumeroTourDragon += 1;
			System.out.println("Vos points de vie : "+J.getPointsDeVieJoueur());
			setAttaqueMonstre(20); 
		}
	}
}

	    	     System.out.println("Vous attaquez le Dragon et infligez 20 dégâts !");

		return NombreDePotionsDisponibles; 
	}

	 public void JoueurPrendrePotion() {
		 int t = this.getNombreDePotionsDisponibles();
		 int points = this.getPointsDeVieJoueur();
		 if (t==2) {
			 if(this.PointsDeVieJoueur == 100) {
				 System.out.println("Votre barre d'énergie est pleine !!");
			 }
			 else { 
			 System.out.println("Vous utilisez une potion.");
			 System.out.println("Vous récupérez 30 points de vie.");
			 t -= 1;
			 points += 30;
			 if (points<=100) {
				 this.setPointsDeVieJoueur(points);
				 this.setNombreDePotionsDisponibles(t);
				 System.out.println("Vos points de vie : "+points+"/100");
				 System.out.println("Potions restantes : 1");
				 System.out.println();
			 }
			 else {
				 this.setPointsDeVieJoueur(100);
				 this.setNombreDePotionsDisponibles(t);
				 System.out.println("Vos points de vie : 100/100");
				 System.out.println("Potions restantes : 1");
				 System.out.println();
			 }
			 }
		 }
		 else {
		 if (t==1) {
			 System.out.println("Vous utilisez votre dernière potion.");
			 System.out.println("Vous récupérez 30 points de vie.");
			 t -= 1;
			 points += 30;
			 if (points<=100) {
				 this.setPointsDeVieJoueur(points);
				 this.setNombreDePotionsDisponibles(t);
				 System.out.println("Vos points de vie : "+points+"/100");
				 System.out.println("Potions restantes : 0");
				 System.out.println();
			 }
			 else {
				 this.setPointsDeVieJoueur(100); 
				 this.setNombreDePotionsDisponibles(t);
				 System.out.println("Vos points de vie : 100/100");
				 System.out.println("Potions restantes : 0");
				 System.out.println();
			 } 
		 }
		 else {
			 System.out.println("ATTENTION : potion non disponible !!!!");
		 }
       }
	 }
	 public void diminuerPointsDeVieJoueur(int i) {
		 int t = this.getPointsDeVieJoueur() - i;
		 if(t>=0)
		 this.setPointsDeVieJoueur(t);
		 else
		 this.setPointsDeVieJoueur(0);
	 }
}
//Classe JeuDonjon dédiée à la gestion du déroulement du jeu
class JeuDonjon{
	private int NumeroSalle;
	private Monstre MonstreEnCours; //monstre en phase de combat
	private boolean JeuContinue; //JeuContinue = True si le jeu continue et false si le jeu se termine
	public JeuDonjon() {
		NumeroSalle = 1;
		JeuContinue = true;
		this.choisirMonstreAleatoire();
	}
	public void transitionSalleSuivante(){
		 if(NumeroSalle != 3) {
			 NumeroSalle += 1;
			 System.out.println("Vous passez à la salle suivante.");
			 System.out.println();
		 }
		 else {
		 NumeroSalle += 1;
		 System.out.println("Vous passez à la dernière salle.");
		 System.out.println();
		 }
	}
	//Choix aléatoire d'un monstre dans chacune des 3 premières salles
	public void choisirMonstreAleatoire() {
		Monstre A = new Gobelin();
		Monstre B = new Orc();
		Monstre[] list_monstres = {A,B};
		int Indice_aléatoirement_choisi = new Random().nextInt(2);
		MonstreEnCours = list_monstres[Indice_aléatoirement_choisi];
	}
	public void setMonstreEnCours(Monstre M) {
		MonstreEnCours = M;
	}
	public boolean getJeuContinue() {
		return JeuContinue;
	}
	public int getNumeroSalle() {
		return NumeroSalle; 
	}
	public Monstre getMonstreEnCours() {
		return MonstreEnCours; 
	}
	public void setJeuContinue(boolean B) {
		JeuContinue = B;
	}
	//Game over automatique si le joueur est forfait
	public void forfait(Joueur J) {
			System.out.println("===================================================");
			System.out.println();
			System.out.println("GAME OVER !");
			System.out.println("Vous êtes déclaré forfait.");
			System.out.println();
			System.out.println("===================================================");
			setJeuContinue(false);
	}
	//Afficher le game over au milieu d'un combat
	public void Game_Over_au_milieu_de_la_partie(Joueur J, Monstre M) {
		if(M.TypeMonstre != "Dragon") {
		
		System.out.println("===================================================");
		System.out.println();
		System.out.println("GAME OVER !");
		System.out.println("Vous avez été vaincu par un "+M.TypeMonstre);
		System.out.println();
		System.out.println("===================================================");
		setJeuContinue(false);
		}
		else {
			System.out.println("===================================================");
			System.out.println();
			System.out.println("GAME OVER !");
			System.out.println("Vous avez été vaincu par le Dragon.");
			System.out.println();
			System.out.println("===================================================");
			setJeuContinue(false);

		}
    }
	//Fin de la partie dans la dernière salle
	public void finPartie(Joueur J) { 
		if(J.getPointsDeVieJoueur() != 0) {
			System.out.println("===================================================");
			System.out.println();
			System.out.println("FÉLICITATIONS !");
			System.out.println("Vous avez vaincu le Dragon.");
			System.out.println();
			System.out.println("===================================================");
			setJeuContinue(false);
		}
	    else {
	     	System.out.println("===================================================");
			System.out.println();
			System.out.println("GAME OVER !");
			System.out.println("Vous avez été vaincu par le Dragon.");
			System.out.println();
			System.out.println("===================================================");
			setJeuContinue(false);
	    }
	}
